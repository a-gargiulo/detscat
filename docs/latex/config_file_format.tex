\hypertarget{config_file_format_autotoc_md1}{}\doxysubsection{\texorpdfstring{Important Notes}{Important Notes}}\label{config_file_format_autotoc_md1}

\begin{DoxyItemize}
\item Arrays are defined using square brackets ({\ttfamily \mbox{[}\mbox{[}...\mbox{]}, ...\mbox{]}}) with comma-\/separated values.
\item Lines starting with {\ttfamily \#} are treated as comments and ignored.
\item All parameter names are case-\/sensitive.
\item Arbitrary whitespace is allowed between keys, values, and delimiters.
\item Blank lines between key-\/value definitions are allowed.
\item Do not enclose strings in {\ttfamily "{}..."{}} or {\ttfamily \textquotesingle{}...\textquotesingle{}}.
\end{DoxyItemize}\hypertarget{config_file_format_autotoc_md2}{}\doxysubsection{\texorpdfstring{Required Parameters}{Required Parameters}}\label{config_file_format_autotoc_md2}
